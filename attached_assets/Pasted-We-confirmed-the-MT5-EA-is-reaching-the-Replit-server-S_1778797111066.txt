Add safe helper functions similar to this MQL5 approach:

string JsonEscape(string value)
{
   StringReplace(value, "\\", "\\\\");
   StringReplace(value, "\"", "\\\"");
   StringReplace(value, "\r", "\\r");
   StringReplace(value, "\n", "\\n");
   StringReplace(value, "\t", "\\t");
   return value;
}

string JString(string value)
{
   return "\"" + JsonEscape(value) + "\"";
}

string JBool(bool value)
{
   return value ? "true" : "false";
}

string JNumber(double value, int digits = 2)
{
   if(!MathIsValidNumber(value))
      return "0.0";

   string output = DoubleToString(value, digits);
   return output;
}

Then build JSON only through helpers, never by raw unescaped string insertion.

Task 2 — Fix heartbeat JSON exactly:
Build heartbeat JSON like this structure with no trailing commas:

string BuildHeartbeatJson()
{
   string account = IntegerToString((long)AccountInfoInteger(ACCOUNT_LOGIN));
   string broker = AccountInfoString(ACCOUNT_COMPANY);
   string server = AccountInfoString(ACCOUNT_SERVER);

   double balance = AccountInfoDouble(ACCOUNT_BALANCE);
   double equity = AccountInfoDouble(ACCOUNT_EQUITY);

   bool liveAllowed =
      (TerminalInfoInteger(TERMINAL_TRADE_ALLOWED) == 1 &&
       MQLInfoInteger(MQL_TRADE_ALLOWED) == 1);

   string json = "{";
   json += "\"account\":" + JString(account) + ",";
   json += "\"broker\":" + JString(broker) + ",";
   json += "\"server\":" + JString(server) + ",";
   json += "\"balance\":" + JNumber(balance, 2) + ",";
   json += "\"equity\":" + JNumber(equity, 2) + ",";
   json += "\"liveAllowed\":" + JBool(liveAllowed);
   json += "}";

   return json;
}

- no trailing comma before }
