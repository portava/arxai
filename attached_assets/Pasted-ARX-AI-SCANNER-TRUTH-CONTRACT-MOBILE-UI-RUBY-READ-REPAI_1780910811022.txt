B. BUILD A SHARED SCANNER TRUTH CONTRACT
============================================================

Create or repair a single shared scanner truth resolver.

  symbolDisplay,
  symbolInternal,
  timeframe,
  quote: {
    bid,
    ask,
    mid,
    source,
    sourceLabel,
    timestamp,
    ageMs,
    status: "live" | "delayed" | "stale" | "unavailable",

    firstTime,
    lastTime,
    lastClose,
    ageMs,

    reason
  },
  consistency: {
    quoteCandlePriceDelta,
    quoteCandlePriceDeltaPct,
    withinTolerance,
    status: "aligned" | "mismatch" | "unknown",
    reason
  },
  permissions: {
    demoManualAllowed,
    demoAIAllowed,
    liveManualAllowed,
    liveAIAllowed,
    sharedBridgeApproved,
    ownBridgeConnected,
    manualTradingBlockedReason,
    liveTradingBlockedReason,
    effectiveMode: "demo" | "live" | "read_only"
  },
  analysis: {
    allowed: boolean,
    level: "full" | "limited" | "historical_only" | "blocked",
    reason
  },
  execution: {
    allowed: boolean,
    reason
  },
  ruby: {

    marketReadAllowed,
